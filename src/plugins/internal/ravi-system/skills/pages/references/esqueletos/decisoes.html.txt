class ExecError extends Error {
  constructor(code, status, body, requestId) {
    super(code);
    Object.assign(this, { code, status, requestId });
    this.fieldErrors = body && typeof body.fieldErrors === "object" ? body.fieldErrors : null;

// POST same-origin { id, input }. 429: espera 2, 4, 8 s e repete o mesmo input (mesma chave).
async function exec(id, input, retries = 3) {
  for (let attempt = 0; ; attempt++) {
    let res;
    try {
      res = await fetch("/_ravi/connectors/exec", { method: "POST", credentials: "same-origin",
        headers: { "content-type": "application/json" }, body: JSON.stringify({ id, input }) });
    } catch { throw new ExecError("network_error", 0, null, null); }
    if (res.status === 429 && attempt < retries) { await new Promise((r) => setTimeout(r, 2000 * 2 ** attempt)); continue; }
    let body = null;
    try { body = await res.json(); } catch { /* não JSON */ }
    if (res.ok && body && body.ok === true) return body.output;
    const code = body && typeof body.error === "string" ? body.error : "connector_unavailable";
    throw new ExecError(code, res.status, body, res.headers.get("x-request-id"));
  }
}

// Sem push: releia com piso de 30 s (limite: 120 chamadas/min por viewer por site).
// Pula a vez com a aba escondida; releitura que dá certo limpa o aviso. Devolve stop().
function poll(fn, ms = 30000) {
  const every = Math.max(ms, 30000);
  let timer = null, stopped = false;
  const tick = async () => {
    if (!document.hidden) { try { await fn(); clearError(); } catch (err) { showError(err); } }
    if (!stopped) timer = setTimeout(tick, every);
  };
  timer = setTimeout(tick, every);
  return () => { stopped = true; clearTimeout(timer); };
}

  });

// Bases: linha da view vira decisão. "opcoes" é JSON em texto (até 16 KB) com { fatos, caminhos };
// "escolha" guarda os ids do caminho ("a/a2") ou "livre"; a aprovação leva a nota com o texto.
// Só a aprovação "approved" conta como decidida: pendente (também depois de um --reset-on) fica aberta.
async function iniciarBases(dados) {
  const C = CONFIG.colunas, versoes = new Map(), nomes = {}, pessoas = {}, chaves = new Map();
  let eu = ""; // id de quem vê: vem na primeira escrita; antes disso, cada decidida mostra o nome
  const view = await describe(CONFIG.viewId);
  for (const col of view.columns || []) {
    const ops = (col.config && col.config.options) || [];
    nomes[col.key] = Object.fromEntries(ops.map((o) => [o.id, o.name]));
  }
  const linha = (r) => {
    const v = r.values || {}, txt = (k) => (typeof v[C[k]] === "string" ? v[C[k]] : "");
    let op = {};
    try { if (txt("opcoes").length <= 16000) op = JSON.parse(txt("opcoes")) || {}; } catch { op = {}; }
    const esc = txt("escolha"), ap = v[C.aprovacao] || {}, resp = txt("resposta"), quando = v[C.pedidaEm];
    const livre = !esc || esc === "livre", por = ap.decidedBy && ap.decidedBy !== eu;
    return {
      id: r.rowId, risco: (nomes[C.risco] || {})[v[C.risco]] || v[C.risco],
      pedidaEm: typeof quando === "string" ? quando : quando && quando.start,
      agente: { nome: txt("agente"), sigla: txt("sigla"), runtime: txt("runtime") },
      pergunta: txt("pergunta"), fatos: op.fatos, caminhos: op.caminhos,
      decidida: ap.state === "approved" ? { caminho: livre ? [] : esc.split("/"), texto: livre ? ap.note : "",
        status: resp ? "feito" : "rodando", resposta: resp, por: por ? (pessoas[ap.decidedBy] || {}).displayName || "Outra pessoa" : "" } : undefined,
    };
  };
  const carregar = async () => {
    const rows = [];
    let cursor;
    do { // segue o nextCursor até 200 linhas
      const out = await query(CONFIG.viewId, { limit: 100, cursor });
      Object.assign(pessoas, out.users || {});
      rows.push(...(out.rows || []));
      cursor = out.nextCursor;
    } while (cursor && rows.length < 200);
    rows.forEach((r) => versoes.set(r.rowId, r.version));
    return { ferramentas: dados.ferramentas, decisoes: rows.map(linha) };
  };
  const ctl = rk.decisions(app, await carregar(), {
    titulo: CONFIG.titulo,
    async onChoose(e) {
      const values = {
        [C.escolha]: e.caminho.length ? e.caminho.join("/") : "livre",
        [C.aprovacao]: { state: "approved", note: (e.texto || e.rotulos.join(" → ")).slice(0, 1000) },
      };
      // "Tentar de novo" com o mesmo input e a mesma versão repete a chave: escrita que passou vira replay.
      const pedido = JSON.stringify([e.decisao, values, versoes.get(e.decisao)]);
      if (!chaves.has(pedido)) chaves.set(pedido, newKey());
      try {
        const r = await update(CONFIG.viewId, e.decisao, values, versoes.get(e.decisao), chaves.get(pedido));
        if (r && r.version) versoes.set(e.decisao, r.version);
        eu = (r && r.values && r.values[C.aprovacao] && r.values[C.aprovacao].decidedBy) || eu;
        clearError();
        ctl.setStatus(e.decisao, "rodando");
      } catch (err) {
        showError(err);
        ctl.setStatus(e.decisao, "erro", MESSAGES[err && err.code] || "Não deu para gravar a escolha. Tente de novo.");
        if (err && err.code === "version_conflict") ctl.set(await carregar());
      }
    },
  });
  poll(async () => ctl.set(await carregar()));
}

// Falha ao abrir: estado de erro com "Tentar de novo"; queda de rede tenta sozinha a cada 30 s.
let espera = 0, abrindo = false;
function abrirBases(dados) {
  if (abrindo) return;
  abrindo = true;
  clearTimeout(espera);
  iniciarBases(dados).catch((err) => {
    const code = err && err.code;
    if (/session|unauthorized/.test(code)) showError(err);
    falha(MESSAGES[code] || "A view de Decisões não respondeu.", () => abrirBases(dados));
    if (/network|unavailable|timeout|rate_limited/.test(code)) espera = setTimeout(() => abrirBases(dados), 30000);
  }).finally(() => { abrindo = false; });
}
