<script>
// CONFIG ----------------------------------------------------------------------
const VIEW_ID = "00000000-0000-0000-0000-000000000000"; // id da view (views create --json)

// LIVRE: textos (o layout form da view, quando existe, tem precedência) -------------
const TEXT = { submit: "Enviar", done: "Enviado. Obrigado!" };

// FIXO --------------------------------------------------------------------------
// person e ref ficam de fora: dono vem de write.set ($viewer.raviUserId).
const SIMPLE = { text: "text", url: "url", email: "email", phone: "tel", number: "number" };
const CHOICE = new Set(["select", "status", "multi_select"]);
let pending = null; // { json, key }: a mesma chave só para repetir o mesmo envio

function field(col) {
  let node;
  if (CHOICE.has(col.type)) {
    node = el("select", { multiple: col.type === "multi_select" });
    if (col.type !== "multi_select") node.append(el("option", { value: "", textContent: "—" }));
    for (const o of col.config.options || []) if (!o.archived) node.append(el("option", { value: o.id, textContent: o.name }));
  } else if (col.type === "checkbox") node = el("input", { type: "checkbox" });
  else if (col.type === "date") node = el("input", { type: col.config.includeTime ? "datetime-local" : "date" });
  else if (col.type === "body") node = el("textarea", { rows: 5 });
  else node = el("input", { type: SIMPLE[col.type] });
  if (col.type === "number") node.step = "any";
  node.name = col.key;
  node.required = col.required && col.type !== "checkbox";
  return node;
}
function read(col, node) {
  if (col.type === "checkbox") return node.checked;
  if (col.type === "multi_select") return [...node.selectedOptions].map((o) => o.value);
  if (node.value === "") return null;
  if (col.type === "number") return Number(node.value);
  if (col.type === "date" && col.config.includeTime) return new Date(node.value).toISOString();
  return node.value; // date sem hora: "YYYY-MM-DD"
}
function build(view) {
  const layout = view.layout && view.layout.type === "form" ? view.layout : {};
  const cols = view.columns.filter((c) => view.capabilities.writeColumns.includes(c.key) && (SIMPLE[c.type] || CHOICE.has(c.type) || ["checkbox", "date", "body"].includes(c.type)));
  const form = document.getElementById("form");
  form.replaceChildren(
    ...(layout.description ? [el("p", { textContent: layout.description })] : []),
    ...cols.map((c) => el("label", {}, c.name + (c.required ? " *" : ""), field(c), el("small", { className: "err", id: `err-${c.key}` }))),
    el("p", { id: "err-form" }),
    el("button", { type: "submit", textContent: layout.submitLabel || TEXT.submit }),
  );
  form.onsubmit = async (event) => {
    event.preventDefault();
    const values = {};
    let body;
    for (const c of cols) {
      const v = read(c, form.elements.namedItem(c.key));
      if (c.type === "body") { if (v !== null) body = v; } else if (v !== null && !(Array.isArray(v) && !v.length)) values[c.key] = v;
    }
    const json = JSON.stringify({ values, body });
    if (!pending || pending.json !== json) pending = { json, key: newKey() };
    form.querySelectorAll("small").forEach((n) => { n.textContent = ""; });
    const status = document.getElementById("err-form");
    const button = form.querySelector("button");
    button.disabled = true;
    try {
      await create(VIEW_ID, values, pending.key, body);
      pending = null;
      form.reset();
      status.className = "ok";
      status.textContent = layout.successMessage || TEXT.done;
      document.getElementById("banner").hidden = true;
    } catch (err) {
      status.className = "err";
      status.textContent = "";
      if (err.code === "validation_failed" && err.fieldErrors) {
        for (const [key, message] of Object.entries(err.fieldErrors)) {
          const slot = document.getElementById(`err-${key}`) || status;
          slot.textContent = slot.textContent ? `${slot.textContent} ${message}` : message;
        }
      } else showError(err);
    } finally {
      button.disabled = false;
    }
  };
  form.hidden = false;
}
async function main() {
  try {
    const view = await describe(VIEW_ID);
    if (!view.valid) throw { code: "view_invalid" };
    if (!view.capabilities.create) throw { code: "base_forbidden" };
    const layout = view.layout && view.layout.type === "form" ? view.layout : {};
    document.getElementById("title").textContent = document.title = layout.title || view.name;
    build(view);
  } catch (err) {
    showError(err);
  }
}
main();
</script>
